# PT-P7-E3-T1 · Record → stop → refine over fixture capture

**Epic:** PT-P7-E3

The end-to-end money shot (PT-P7-R4): a recording started by clicking the real UI runs the real
orchestrator → engine subprocess → `live.md` → refinement queue → `final.md`, from the committed
paired fixture WAVs (PT-P7-R2), asserting artifacts and event order. Needs the live + refine models:
the launch adds `PULSARTRACE_MODELS_DIR` pointing at the host's real cache (the explicit read-only
share, PT-P7-R9/PT-P7-D3) — on a fresh host, run any model-downloading filter once first
(`swift test --filter Parakeet`, `--filter WhisperKitRefine`).

## Acceptance criteria

- The test starts recording via `pt.menubar.recordToggle`, observes `live.md` appear and grow in the
  seeded output root, and needs no stop click — the engine self-exits at fixture EOF (~16 s of
  paired audio) and the app finalizes (PT-P7-E1-T6).
- It then waits (generous timeout, ≤ 300 s) for `final.md` to replace `live.md`, and asserts: the
  `<!-- pulsartrace:final -->` marker, at least one `**[` utterance line, and a non-empty
  `metadata.json`.
- Event order in `<home>/Library/Application Support/PulsarTrace/events/*.jsonl`: the live-start
  event precedes the refinement events, refinement start precedes completion. Pin the exact `type`
  strings from `.erratum/product/architecture/events-log.md` (PT-C6) when writing the test — do not
  guess them; the strings below are written from that contract's vocabulary and must be corrected
  against it if they differ.

```swift
import XCTest

/// Full record → refine pass over fixture capture (PT-P7-R4, PT-P7-R2).
final class RecordFlowTests: XCTestCase {

    /// Repo-root-relative fixture paths, derived from #filePath
    /// (UITests/PulsarTraceUITests/RecordFlowTests.swift → repo root).
    private static let repoRoot = URL(fileURLWithPath: #filePath)
        .deletingLastPathComponent()   // PulsarTraceUITests
        .deletingLastPathComponent()   // UITests
        .deletingLastPathComponent()   // repo root
    private static let pairedDir = repoRoot
        .appendingPathComponent("Tests/Fixtures/audio/mic-and-system-paired")

    func testFixtureRecordingRefinesToFinalMarkdown() async throws {
        let seed = try await SeededHome.make()
        defer { seed.tearDown() }

        let app = XCUIApplication()
        app.launchEnvironment.merge(
            seed.launchEnvironment, uniquingKeysWith: { _, new in new })
        app.launchEnvironment["PULSARTRACE_SYSTEM_FIXTURE"] =
            Self.pairedDir.appendingPathComponent("system.wav").path
        app.launchEnvironment["PULSARTRACE_MIC_FIXTURE"] =
            Self.pairedDir.appendingPathComponent("mic.wav").path
        // Explicit read-only model share (PT-P7-D3) — live + refine models.
        app.launchEnvironment["PULSARTRACE_MODELS_DIR"] =
            FileManager.default.homeDirectoryForCurrentUser
                .appendingPathComponent("Library/Caches/PulsarTrace/models").path
        app.launch()
        defer { app.terminate() }

        // Start from the panel.
        let byId = app.statusItems["pt.statusItem"]
        let item = byId.exists ? byId : app.statusItems.firstMatch
        XCTAssertTrue(item.waitForExistence(timeout: 15))
        item.click()
        let record = app.buttons["pt.menubar.recordToggle"]
        XCTAssertTrue(record.waitForExistence(timeout: 5))
        record.click()

        // A new recording folder appears; live.md exists and grows.
        let newFolder = try poll(timeout: 60, message: "recording folder") {
            try newestFolder(in: seed.outputRoot,
                             excluding: SeededHome.recordingFolders)
        }
        let liveURL = newFolder.appendingPathComponent("live.md")
        _ = try poll(timeout: 60, message: "live.md") {
            FileManager.default.fileExists(atPath: liveURL.path) ? true : nil
        }
        let sizeA = try size(of: liveURL)
        _ = try poll(timeout: 60, message: "live.md growth") {
            (try size(of: liveURL)) > sizeA ? true : nil
        }

        // Engine self-exits at fixture EOF; refinement replaces live → final.
        let finalURL = newFolder.appendingPathComponent("final.md")
        _ = try poll(timeout: 300, message: "final.md") {
            FileManager.default.fileExists(atPath: finalURL.path) ? true : nil
        }
        let final = try String(contentsOf: finalURL, encoding: .utf8)
        XCTAssertTrue(final.hasPrefix("<!-- pulsartrace:final -->"))
        XCTAssertTrue(final.contains("**["), "no utterance line in final.md")
        let metadata = newFolder.appendingPathComponent("metadata.json")
        XCTAssertGreaterThan(try size(of: metadata), 0)

        // Event causal order (PT-C6). PIN THESE STRINGS against
        // .erratum/product/architecture/events-log.md before merging.
        let types = try eventTypes(home: seed.home)
        let liveIdx = types.firstIndex { $0.hasPrefix("live_md") }
        let refineStartIdx = types.firstIndex { $0 == "refinement_started" }
        let refineDoneIdx = types.firstIndex { $0 == "refinement_completed" }
        XCTAssertNotNil(liveIdx); XCTAssertNotNil(refineStartIdx)
        XCTAssertNotNil(refineDoneIdx)
        XCTAssertLessThan(liveIdx!, refineStartIdx!)
        XCTAssertLessThan(refineStartIdx!, refineDoneIdx!)
    }

    // MARK: - Helpers

    private func poll<T>(
        timeout: TimeInterval, message: String,
        _ probe: () throws -> T?
    ) throws -> T {
        let deadline = Date().addingTimeInterval(timeout)
        while Date() < deadline {
            if let value = try probe() { return value }
            Thread.sleep(forTimeInterval: 0.5)
        }
        XCTFail("timed out waiting for \(message)")
        throw XCTSkip("unreachable")
    }

    private func newestFolder(in root: URL, excluding: [String]) throws -> URL? {
        let names = try FileManager.default.contentsOfDirectory(atPath: root.path)
        return names.filter { !excluding.contains($0) }.sorted().last
            .map { root.appendingPathComponent($0, isDirectory: true) }
    }

    private func size(of url: URL) throws -> Int {
        (try FileManager.default.attributesOfItem(atPath: url.path)[.size]
            as? Int) ?? 0
    }

    private func eventTypes(home: URL) throws -> [String] {
        let dir = home.appendingPathComponent(
            "Library/Application Support/PulsarTrace/events")
        let files = (try? FileManager.default
            .contentsOfDirectory(atPath: dir.path)) ?? []
        return try files.sorted().flatMap { file -> [String] in
            try String(contentsOf: dir.appendingPathComponent(file),
                       encoding: .utf8)
                .split(separator: "\n")
                .compactMap { line in
                    (try? JSONSerialization.jsonObject(
                        with: Data(line.utf8)) as? [String: Any])?["type"]
                        as? String
                }
        }
    }
}
```
