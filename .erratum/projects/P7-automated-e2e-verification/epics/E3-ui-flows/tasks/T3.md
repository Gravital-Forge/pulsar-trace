# PT-P7-E3-T3 · Speaker rename flow with retroactive rewrite assertions

**Epic:** PT-P7-E3

Rename through the real Speakers pane against the pre-seeded state (PT-P7-D6) and assert what the
smoke checklist checks by eye: every past `final.md` rewritten, `.bak` siblings left, cause event
before effect events (Hard Invariant #8).

## Acceptance criteria

- Renaming Alice → Alicia via `pt.speakers.renameField` updates both seeded `final.md` files on disk
  ("Alicia" present, standalone "Alice" label gone), leaves a `.bak` in each folder, and the
  Speakers pane shows the new name.
- The isolated events log contains `speaker_renamed` before its `final_md_rewritten` events (two,
  one per rewritten recording).

```swift
import XCTest

/// Speaker-management flows over pre-seeded state (PT-P7-R4, PT-P7-D6).
final class SpeakerFlowTests: XCTestCase {

    private var seed: SeededHome!
    private var app: XCUIApplication!

    override func setUp() async throws {
        continueAfterFailure = false
        seed = try await SeededHome.make()
        app = XCUIApplication()
        app.launchEnvironment.merge(
            seed.launchEnvironment, uniquingKeysWith: { _, new in new })
        app.launch()
        openSpeakersPane()
    }

    override func tearDown() {
        app.terminate()
        seed.tearDown()
    }

    func testRenameRewritesAllPastFinalMarkdown() throws {
        let aliceId = seed.speakerIds["Alice"]!
        let row = app.descendants(matching: .any)["pt.speakers.row.\(aliceId)"]
        XCTAssertTrue(row.waitForExistence(timeout: 10))
        row.click()

        let field = app.descendants(matching: .any)["pt.speakers.renameField"]
        XCTAssertTrue(field.waitForExistence(timeout: 10))
        field.click()
        field.typeKey("a", modifierFlags: .command)   // select-all
        field.typeText("Alicia\r")

        // The rewrite runs briefly; poll the artifacts, not the spinner.
        for folder in SeededHome.recordingFolders {
            let finalURL = seed.outputRoot
                .appendingPathComponent(folder)
                .appendingPathComponent("final.md")
            try poll(timeout: 30, message: "rewrite of \(folder)") {
                let text = try String(contentsOf: finalURL, encoding: .utf8)
                return text.contains("Alicia:**") ? true : nil
            }
            let text = try String(contentsOf: finalURL, encoding: .utf8)
            XCTAssertFalse(text.contains("Alice:**"),
                           "\(folder) still carries the old label")
            let bak = (try FileManager.default.contentsOfDirectory(
                atPath: seed.outputRoot.appendingPathComponent(folder).path))
                .contains { $0.hasSuffix(".bak") }
            XCTAssertTrue(bak, "no .bak left in \(folder)")
        }

        // Cause before effect (Hard Invariant #8): speaker_renamed precedes
        // its two final_md_rewritten events.
        let types = try eventTypes(home: seed.home)
        guard let renameIdx = types.firstIndex(of: "speaker_renamed") else {
            return XCTFail("no speaker_renamed event")
        }
        let rewrites = types.indices.filter {
            types[$0] == "final_md_rewritten" && $0 > renameIdx
        }
        XCTAssertGreaterThanOrEqual(rewrites.count, 2,
            "expected a final_md_rewritten per rewritten recording, after the cause")

        // The pane reflects the rename.
        XCTAssertTrue(app.descendants(matching: .any)["pt.speakers.row.\(aliceId)"]
            .staticTexts["Alicia"].waitForExistence(timeout: 10))
    }

    // MARK: - Helpers (shared with T4's test in this same file)

    func openSpeakersPane() {
        let byId = app.statusItems["pt.statusItem"]
        let item = byId.exists ? byId : app.statusItems.firstMatch
        XCTAssertTrue(item.waitForExistence(timeout: 15))
        item.click()
        app.buttons["pt.menubar.openMainWindow"].click()
        XCTAssertTrue(app.windows.firstMatch.waitForExistence(timeout: 10))
        app.buttons["pt.sidebar.speakers"].click()
    }

    @discardableResult
    func poll<T>(timeout: TimeInterval, message: String,
                 _ probe: () throws -> T?) throws -> T {
        let deadline = Date().addingTimeInterval(timeout)
        while Date() < deadline {
            if let value = try probe() { return value }
            Thread.sleep(forTimeInterval: 0.5)
        }
        XCTFail("timed out waiting for \(message)")
        throw XCTSkip("unreachable")
    }

    func eventTypes(home: URL) throws -> [String] {
        let dir = home.appendingPathComponent(
            "Library/Application Support/PulsarTrace/events")
        let files = (try? FileManager.default
            .contentsOfDirectory(atPath: dir.path)) ?? []
        return try files.sorted().flatMap { file -> [String] in
            try String(contentsOf: dir.appendingPathComponent(file),
                       encoding: .utf8)
                .split(separator: "\n")
                .compactMap { line in
                    (try? JSONSerialization.jsonObject(
                        with: Data(line.utf8)) as? [String: Any])?["type"]
                        as? String
                }
        }
    }
}
```
