# PT-P7-E3-T2 · Settings persistence across relaunch

**Epic:** PT-P7-E3

The smoke checklist's "settings persist across relaunch" item as a deterministic test (PT-P7-R4):
change a setting through the real Settings pane, kill the app, relaunch on the same isolated suite,
and read the value back through the UI.

## Acceptance criteria

- Toggling `pt.settings.systemAudioToggle` off, terminating, and relaunching with the same
  `SeededHome` environment shows the toggle still off.
- The seeded output folder still renders after relaunch (the path survived, no re-default).
- The test never touches the production suite (it runs entirely on the seeded suite).

## Files

- Create: `UITests/PulsarTraceUITests/SettingsPersistenceTests.swift`

- [ ] **Step 1: Write the test**

```swift
import XCTest

/// Settings persist across an app relaunch on the same isolated suite
/// (PT-P7-R4; smoke-checklist "Settings persistence").
final class SettingsPersistenceTests: XCTestCase {

    func testSystemAudioToggleSurvivesRelaunch() async throws {
        let seed = try await SeededHome.make()
        defer { seed.tearDown() }

        var app = launch(seed)
        openSettings(app)
        let toggle = app.descendants(matching: .any)["pt.settings.systemAudioToggle"]
        XCTAssertTrue(toggle.waitForExistence(timeout: 10))
        XCTAssertEqual(toggleState(toggle), true, "seeded default is on")
        toggle.click()
        XCTAssertEqual(toggleState(toggle), false)

        app.terminate()
        app = launch(seed)
        openSettings(app)
        let after = app.descendants(matching: .any)["pt.settings.systemAudioToggle"]
        XCTAssertTrue(after.waitForExistence(timeout: 10))
        XCTAssertEqual(toggleState(after), false,
                       "toggle did not survive the relaunch")

        let field = app.descendants(matching: .any)["pt.settings.outputFolderField"]
        XCTAssertTrue(field.waitForExistence(timeout: 10))
        XCTAssertTrue((field.value as? String)?
            .contains(seed.outputRoot.lastPathComponent) == true
            || field.staticTexts[seed.outputRoot.path].exists,
            "seeded output folder lost across relaunch")
        app.terminate()
    }

    // MARK: - Helpers

    private func launch(_ seed: SeededHome) -> XCUIApplication {
        let app = XCUIApplication()
        app.launchEnvironment.merge(
            seed.launchEnvironment, uniquingKeysWith: { _, new in new })
        app.launch()
        return app
    }

    private func openSettings(_ app: XCUIApplication) {
        let byId = app.statusItems["pt.statusItem"]
        let item = byId.exists ? byId : app.statusItems.firstMatch
        XCTAssertTrue(item.waitForExistence(timeout: 15))
        item.click()
        app.buttons["pt.menubar.openMainWindow"].click()
        XCTAssertTrue(app.windows.firstMatch.waitForExistence(timeout: 10))
        app.buttons["pt.sidebar.settings"].click()
    }

    /// SwiftUI toggles surface state as value "1"/"0" (or Bool on newer OSes).
    private func toggleState(_ element: XCUIElement) -> Bool? {
        if let s = element.value as? String { return s == "1" }
        return element.value as? Bool
    }
}
```
