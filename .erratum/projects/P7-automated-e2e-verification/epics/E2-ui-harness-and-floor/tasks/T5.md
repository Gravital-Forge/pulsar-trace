# PT-P7-E2-T5 · The floor suite

**Epic:** PT-P7-E2

The user-set gate of PT-P7-R4: launch on a `SeededHome`, open every **idle-reachable** surface —
menubar panel, Recordings, Speakers, Settings — and verify the seeded data renders in each; plus the
PT-P7-R9 proof that a run leaves the daily instance's real state untouched. (Scope amended
2026-07-03 per PT-P7-D8: the live-transcript window is recording-only and moved to the PT-P7-E3
record flow; the `testLiveTranscriptWindowOpens` snippet below is superseded.) Everything is located
by `A11yID` — a failure here means an identifier is missing or a surface regressed, which is exactly
the class of bug this project exists to catch.

## Acceptance criteria

- `FloorTests` passes via `scripts/run-ui-tests.sh -only-testing:PulsarTraceUITests/FloorTests`.
- Each pane assertion is content-bearing (a seeded folder row, seeded speaker names, the seeded
  output-folder path), not mere existence of the container.
- Daily-state check: for every real path that exists (`~/Library/Application Support/PulsarTrace`,
  `~/Library/Logs/PulsarTrace`, the production defaults plist), the modification time is unchanged
  across the run.

- [ ] **Step 1: Write the suite**

```swift
import XCTest

/// The PT-P7-R4 floor: every surface opens and renders the seeded state;
/// the run never touches daily state (PT-P7-R9).
final class FloorTests: XCTestCase {

    private var seed: SeededHome!
    private var app: XCUIApplication!
    /// Real daily-state paths captured before launch → mtime.
    private var dailyState: [String: Date] = [:]

    private var realStatePaths: [String] {
        let home = FileManager.default.homeDirectoryForCurrentUser.path
        return [
            "\(home)/Library/Application Support/PulsarTrace",
            "\(home)/Library/Logs/PulsarTrace",
            "\(home)/Library/Preferences/com.gravitalforge.PulsarTrace.plist",
        ]
    }

    override func setUp() async throws {
        continueAfterFailure = false
        for path in realStatePaths {
            if let attrs = try? FileManager.default
                .attributesOfItem(atPath: path),
               let mtime = attrs[.modificationDate] as? Date {
                dailyState[path] = mtime
            }
        }
        seed = try await SeededHome.make()
        app = XCUIApplication()
        app.launchEnvironment.merge(
            seed.launchEnvironment, uniquingKeysWith: { _, new in new })
        app.launch()
    }

    override func tearDown() async throws {
        app.terminate()
        // PT-P7-R9: the run must not have touched any real daily-state path.
        for (path, before) in dailyState {
            let after = (try? FileManager.default
                .attributesOfItem(atPath: path))?[.modificationDate] as? Date
            XCTAssertEqual(before, after,
                           "daily state was modified: \(path)")
        }
        seed.tearDown()
    }

    private func openPanel() {
        let byId = app.statusItems["pt.statusItem"]
        let item = byId.exists ? byId : app.statusItems.firstMatch
        XCTAssertTrue(item.waitForExistence(timeout: 15))
        item.click()
    }

    private func openMainWindow() {
        openPanel()
        app.buttons["pt.menubar.openMainWindow"].click()
        XCTAssertTrue(app.windows.firstMatch.waitForExistence(timeout: 10))
    }

    func testMenubarPanelShowsRecordControl() {
        openPanel()
        XCTAssertTrue(app.buttons["pt.menubar.recordToggle"]
            .waitForExistence(timeout: 5))
    }

    func testRecordingsPaneRendersSeededRecordings() {
        openMainWindow()
        app.buttons["pt.sidebar.recordings"].click()
        for folder in SeededHome.recordingFolders {
            XCTAssertTrue(
                app.descendants(matching: .any)["pt.recordings.row.\(folder)"]
                    .waitForExistence(timeout: 10),
                "seeded recording \(folder) not rendered")
        }
    }

    func testSpeakersPaneRendersSeededLibrary() {
        openMainWindow()
        app.buttons["pt.sidebar.speakers"].click()
        for (name, id) in seed.speakerIds {
            let row = app.descendants(matching: .any)["pt.speakers.row.\(id)"]
            XCTAssertTrue(row.waitForExistence(timeout: 10),
                          "seeded speaker \(name) not rendered")
            XCTAssertTrue(row.staticTexts[name].exists,
                          "row for \(name) does not show the name")
        }
    }

    func testSettingsPaneRendersSeededValues() {
        openMainWindow()
        app.buttons["pt.sidebar.settings"].click()
        for id in ["pt.settings.micPicker", "pt.settings.refineModelPicker",
                   "pt.settings.outputFolderField",
                   "pt.settings.systemAudioToggle",
                   "pt.settings.hotkeyRecorder", "pt.settings.mcpToggle"] {
            XCTAssertTrue(app.descendants(matching: .any)[id]
                .waitForExistence(timeout: 10), "missing control: \(id)")
        }
        // Content-bearing: the field shows the seeded output folder.
        let field = app.descendants(matching: .any)["pt.settings.outputFolderField"]
        XCTAssertTrue((field.value as? String)?
            .contains(seed.outputRoot.lastPathComponent) == true
            || field.staticTexts[seed.outputRoot.path].exists,
            "seeded output folder not shown")
    }

    func testLiveTranscriptWindowOpens() {
        openPanel()
        app.buttons["pt.menubar.openLiveTranscript"].click()
        XCTAssertTrue(app.descendants(matching: .any)["pt.liveTranscript.window"]
            .waitForExistence(timeout: 10)
            || app.windows["Live Transcript"].waitForExistence(timeout: 5))
    }
}
```

Query-shape note: SwiftUI surfaces containers inconsistently across element types, which is why
row/container lookups use `descendants(matching: .any)[identifier]`. Where a lookup fails on the
current macOS, adjust the **query shape** (element type), never the locator strategy — locators stay
identifiers (PT-P7-R3).

Two writing rules for the real code (the snippets bend them for readability): reference identifiers
through the `A11yID` constants (`import PulsarTraceMenuBar`) rather than string literals — the
compile-time guarantee is why the constants exist (E2-T1) — and if the daily app is running during a
local run, its own settings writes can legitimately touch the production preferences plist; quit it
for this suite, or scope the mtime check to the two directories when that proves noisy.
